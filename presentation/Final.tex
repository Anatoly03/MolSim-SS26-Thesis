\subsection{The Ownership Model}
\simpleframe\begin{frame}[t]{The Ownership Model}{Ownership}
    Rust guarantees memory safety and automatically frees memory when a value goes out of scope. This is achieved through the ``Ownership model'', a set of rules which are enforced at compile time.

    {\centering
    \begin{tikzpicture}[node distance=1cm and 1cm, minimum height=1cm, minimum width=3cm]
        \node[draw, rectangle] (owned) {Owned};
        \node[draw, rectangle, gray!30, right=of owned] (mutborrow) {\gray{\texttt{\&mut}}};
        \node[draw, rectangle, gray!30, right=of mutborrow] (borrow) {\gray{\texttt{\&}}};

        \draw[->, gray!30, thick] (owned) -- (mutborrow) node[midway, above] {};
        \draw[->, gray!30, thick] (mutborrow) -- (borrow) node[midway, above] {};
        \draw[->, gray!30, thick] (mutborrow) to[bend left = 10] (borrow) node[midway, above] {};
        \draw[->, gray!30, thick] (mutborrow) to[bend right = 10] (borrow) node[midway, above] {};
        \draw[->, gray!30, thick] (owned) to[bend left = 15] (borrow) node[midway, above] {};
    \end{tikzpicture}
    \vspace{1em}

    \begin{description}
        \item[\blue{Rule 1}] The value has an owner,
        \item[\blue{Rule 2}] and only one owner at any time.
        \item[\blue{Rule 3}]  When the owner leaves the scope, without passing on ownership, the value will be dropped.
    \end{description}}

    \blue{Intuition:} The scope of the owner defines the \blue{lifetime} of a value.
\end{frame}

\simpleframe\begin{frame}[t,fragile]{The Ownership Model}{Ownership}
    \begin{lstlisting}[frame=single,breaklines=false]
fn problematic_move() {
    let s = String::from("hello");
    {                // inner scope
        let t = s;   // s is moved to t
    }                // t goes out of scope
    println!("{s}"); // error: use of moved value
}
    \end{lstlisting}
    \vspace{0.5em}

    This example breaks \blue{Rule 2: There can only be one owner}. When \texttt{b} goes out of scope, the value is freed.
\end{frame}

\simpleframe\begin{frame}[t]{The Ownership Model}{Borrowing and References}
    To make sure values can be shared across different scopes, Rust introduces the concept of \blue{borrowing}: When references go out of scope, the value is not dropped, and a reference cannot outlive its value.

    {\centering
    \begin{tikzpicture}[node distance=1cm and 1cm, minimum height=1cm, minimum width=3cm]
        \node[draw, rectangle] (owned) {Owned};
        \node[draw, rectangle, right=of owned] (mutborrow) {\texttt{\&mut}};
        \node[draw, rectangle, right=of mutborrow] (borrow) {\texttt{\&}};

        \draw[->, thick] (owned) -- (mutborrow) node[midway, above] {};
        \draw[->, thick] (mutborrow) -- (borrow) node[midway, above] {};
        \draw[->, thick] (mutborrow) to[bend left = 10] (borrow) node[midway, above] {};
        \draw[->, thick] (mutborrow) to[bend right = 10] (borrow) node[midway, above] {};
        \draw[->, thick] (owned) to[bend left = 15] (borrow) node[midway, above] {};
    \end{tikzpicture}
    \vspace{1em}

    \begin{description}
        \item[\blue{Rule 4}] At any given time, an item can either have one mutable reference or multiple immutable references,
        \item[\blue{Rule 5}] and any reference must be valid (no dangling references!).
    \end{description}}

    \vspace{1.5em}
    \blue{Intuition:} Owner can read, write and free. Mutable borrow can read and write. Immutable borrow can only read.
\end{frame}

\simpleframe\begin{frame}[t,fragile]{The Ownership Model}{Borrowing and References}
    \begin{lstlisting}[frame=single,breaklines=false]
fn too_many_references() {
    let s = String::from("hello");
    let muts = &mut s; // 'muts' is a mutable borrow of 's'
    println!("{s}");   // problem: printing a value borrows it
                       // immutable (reading value content)
                       // 'muts' out of scope
}
    \end{lstlisting}
    \vspace{0.5em}

    This example breaks \blue{Rule 4: Either one mutable reference or multiple immutable}.
\end{frame}

\simpleframe\begin{frame}[t,fragile]{The Ownership Model}{Borrowing and References}
    \begin{lstlisting}[frame=single,breaklines=false]
fn dangle() -> &String {
    let s = String::from("hello");
    return &s; // returns a reference to s;
               // s goes out of scope => &s is invalid
}
    \end{lstlisting}
    \vspace{0.5em}

    This example breaks \blue{Rule 5: Any reference must be valid}.

\subsection{Traits vs Interfaces}
\simpleframe\begin{frame}[fragile]{Traits vs Interfaces}
    Rust has no inheritance. Instead, shared behaviour is expressed through \blue{traits}, an abstract interface that a type can implement \blue{explicitly}.

    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Definition}
        \begin{lstlisting}[language=Rust,basicstyle=\ttfamily\scriptsize]
pub trait Potential {
    fn potential(
        &self,
        a: &Particle,
        b: &Particle
    ) -> Vec3;
}
        \end{lstlisting}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Implementation for a type}
        \begin{lstlisting}[language=Rust,basicstyle=\ttfamily\scriptsize]
impl Potential for NewtonForce {
    fn potential(...) -> f64 {
        let d = Particle::dist(a, b);
        let m = Particle::mass_prod(a, b);
        -self.g * m / d
    }
}
        \end{lstlisting}
        \end{column}
    \end{columns}

    \vspace{0.5em}
    In C++ a struct can inherit only one parent struct, but in Rust a type can \blue{implement multiple traits}. In code we have a \texttt{ParticleContainer} trait which the three approaches implement.
\end{frame}

\subsection{Dynamic Dispatch}
\simpleframe\begin{frame}[fragile]{Dynamic Dispatch}
    For the simulation, the concrete force model and particle containers are chosen at runtime. For C++ we achieve this with virtual functions and in Rust we use \blue{dynamic dispatch}.

    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Rust: Type coercion with \texttt{dyn}}
\begin{lstlisting}[language=Rust,basicstyle=\ttfamily\scriptsize]
let newton = NewtonForce { g: 1.0 };

let force: &dyn Force = &newton;


println!("{}", force
    .potential(&p1, &p2));
\end{lstlisting}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{C++: \texttt{virtual} functions}
\begin{lstlisting}[language=C++,basicstyle=\ttfamily\scriptsize]
auto newton = std::make_unique<NewtonForce>(1.0);
std::unique_ptr<Force> force = std::move(newton);

std::cout << force
    ->potential(p1, p2);
\end{lstlisting}
        \end{column}
    \end{columns}

    \vspace{0.5em}
    Both Rust and C++ make an indirect method call, which is \blue{not a zero-cost abstraction!}
\end{frame}

\subsection{Direct Sum}
\nofooterframe\begin{frame}[fragile]{Direct Sum}{Conceptual Implementation\footnote{The code snippet is in Rust-style pseudo code. You cannot implicitly cast an \texttt{Option<T>}\newline to \texttt{T} in practice.}}
    The \text{DirectSum} algorithm uses the naive approach: Two loops are used to compute all particle pair indeces \texttt{i} and \texttt{j}.

    \vspace{0.5em}
\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
let p0 = self.particles.get_mut(i);
let p1 = self.particles.get_mut(j); // problem: rule 4

callback(&mut p0, &p1);
callback(&mut p1, &p0);
\end{lstlisting}
    \vspace{0.5em}

    This breaks \blue{Rule 4}: The signature of \texttt{Vec::get\_mut} \blue{borrows the entire vector mutably!}

    \vspace{0.5em}
\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
pub fn get_mut(&mut self, index: usize) -> Option<&mut Particle>;
\end{lstlisting}
    \vspace{0.5em}
\end{frame}

\simpleframe\begin{frame}[fragile]{Direct Sum}{The Solution}
    The solution is provided by the Rust standard library: Similar to placing an \blue{invariant}, you can \blue{split the vector logically into two mutable arrays:}

    \vspace{0.5em}
\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
pub fn split_at_mut(&mut self, middle: usize)
    -> (&mut [Particle], &mut [Particle]);
\end{lstlisting}
    \vspace{0.5em}

    This gives us the following code snippet for the pairwise for loop over indeces \texttt{i} and \texttt{j}:

    \vspace{0.5em}
\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
let (left, right) = self.particles.split_at_mut(j);

let p0 = left.get_mut(i); // borrows left mutably
let p1 = right.get_mut(j); // borrows right mutably

callback(&mut p0, &p1);
callback(&mut p1, &p0);
\end{lstlisting}
\end{frame}

\subsection{Linked Cells}
\nofooterframe\begin{frame}[fragile]{Linked Cells}
    The \texttt{LinkedCells} approach partitions the 3D space into \blue{cells}, reducing pair computation to within a cell or inbetween neighbouring cells. The cell type is a generic\footnote{Base support for algorithms like \texttt{LinkedCells<ParallelDirectSum>} or  \texttt{LinkedCells<LinkedCells<DirectSum>>} exists but was never tested or benchmarked.} which implements \texttt{ParticleContainer}.

    \vspace{1em}
\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
pub struct LinkedCells<Cell: ParticleContainer> {
    cells: HashMap<Vec3<i32>, Cell>,
    cell_size: Vec3<f64>,
}
\end{lstlisting}
    \vspace{0.5em}


\end{frame}

\subsection{Parallel}
\simpleframe\begin{frame}[fragile]{Parallel}
    For the parallelization approach, we parallelized the outer loop. In Rust, we had to disable memory-safety mechanisms using \texttt{unsafe} to get this to work.

    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Rayon}
\begin{lstlisting}[language=Rust,basicstyle=\ttfamily\scriptsize]
(0..count).into_par_iter().for_each(|i| unsafe {
  let mut p1 = &mut *particles.at(i);

  for j in 0 ..count {
    let right = &*particles.at(j);
    callback(&mut p0, &p1);
  }
});
\end{lstlisting}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{OpenMP}
\begin{lstlisting}[language=C++,basicstyle=\ttfamily\scriptsize]
#pragma omp parallel for
for (size_t i = 0; i < particles_vec.size(); ++i) {
  const auto& p1 = particles_vec[i];

  for (size_t j = 0; j < particles_vec.size(); ++j) {
    const auto& p2 = particles_vec[j];
    callback(p1, p2);
  }
}

\simpleframe\begin{frame}{Ownership and Borrowing: Rules}
    \begin{enumerate}
        \item The value has an owner,
        \item and only one owner at any time.
        \item  When the owner leaves the scope, without passing on ownership, the value will be dropped.
        \item At any given time, an item can either have one mutable reference or multiple immutable references,
        \item and any reference must be valid (no dangling references!).
    \end{enumerate}
\end{frame}

\simpleframe\begin{frame}[fragile]{Direct Sum}{Conceptual Implementation\footnote{The code snippet is in Rust-style pseudo code. You cannot implicitly cast an \texttt{Option<T>}\newline to \texttt{T} in practice.}}

\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
type Callback = Fn(&mut Particle, &Particle);
// self.particles is a vector of particles

fn for_each_pairs(&mut self, callback: &dyn Callback) {
    let count = self.particles.len();

    for i in 0..count {
        for j in (i+1) ..count {
            let p0 = self.particles.get_mut(i); // borrows self.particles
            let p1 = self.particles.get_mut(j); // problem: rule 4

            callback(&mut p0, &p1);
            callback(&mut p1, &p0);
        }
    }
}
\end{lstlisting}
\end{frame}

\simpleframe\begin{frame}[fragile]{Direct Sum}{The Solution}
\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
fn for_each_pairs(&mut self, callback: &dyn Callback) {
    let count = self.particles.len();

    for i in 0..count {
        for j in (i+1) ..count {
            // next line borrows self.particles.
            let (left, right) = self.particles.split_at_mut(j);

            let p0 = left.get_mut(i); // borrows left mutably
            let p1 = right.get_mut(j); // borrows right mutably

            callback(&mut p0, &p1);
            callback(&mut p1, &p0);
        }
    }
}
\end{lstlisting}
\end{frame}

\simpleframe\begin{frame}[fragile]{Parallel Direct Sum}{Rust}
\begin{lstlisting}[frame=single,breaklines=false,language=Rust,basicstyle=\ttfamily\scriptsize]
type Callback = Fn(&mut Particle, &Particle);

fn for_each_pairs(&mut self, callback: &dyn Callback) {
    let count = self.particles.len();

    (0..count).into_par_iter().for_each(|i| unsafe {
        let mut p1 = &mut *particles.at(i);

        for j in 0 ..count {
            let right = &*particles.at(j);
            callback(&mut p0, &p1);
        }
    });
}
\end{lstlisting}
\end{frame}

\simpleframe\begin{frame}[fragile]{Parallel Direct Sum}{C++}
\begin{lstlisting}[frame=single,breaklines=false,language=C++,basicstyle=\ttfamily\scriptsize]
typedef Callback std::function<void(const Particle &, const Particle &)>;

void for_each_pairs(Callback callback) const override {
#pragma omp parallel for
    for (size_t i = 0; i < particles_vec.size(); ++i) {
        const auto& p1 = particles_vec[i];

        for (size_t j = 0; j < particles_vec.size(); ++j) {
            if (i == j) continue; // skip self-interaction
            const auto& p2 = particles_vec[j];
            callback(p1, p2);
        }
    }
}
\end{lstlisting}
\end{frame}
